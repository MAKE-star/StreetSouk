// lib/shopify.ts — Shopify Storefront API (Headless)
// Demonstrates GraphQL query structure for product variants + pricing

const SHOPIFY_STOREFRONT_URL = `https://streetsouk.myshopify.com/api/2024-01/graphql.json`;

export const GET_PRODUCT_DROP = `
  query getLatestDrop($handle: String!) {
    product(handle: $handle) {
      id
      title
      description
      tags
      images(first: 3) {
        edges { node { url altText } }
      }
      variants(first: 10) {
        edges {
          node {
            id
            title
            availableForSale
            price { amount currencyCode }
            selectedOptions { name value }
          }
        }
      }
    }
  }
`;

// API call with access token — this returns what you see rendered above
export async function fetchProduct(handle: string) {
  const res = await fetch(SHOPIFY_STOREFRONT_URL, {
    method: 'POST',
    headers: {
      'Content-Type': 'application/json',
      'X-Shopify-Storefront-Access-Token': process.env.SHOPIFY_TOKEN!
    },
    body: JSON.stringify({
      query: GET_PRODUCT_DROP,
      variables: { handle }
    })
  });
  return (await res.json()).data.product;
}

    }]);
